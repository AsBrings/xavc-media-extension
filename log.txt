// FNV-1a 64 over the luma plane: hash10 over the native samples as 16-bit values, hash8 over
// the samples rounded to 8 bits exactly like the decoder MFT does for NV12 / YUY2 output.
static void luma_hashes(const AVFrame* fr, unsigned long long* h10, unsigned long long* h8) {
    const AVPixFmtDescriptor* d = av_pix_fmt_desc_get(fr->format);
    int depth = d->comp[0].depth;
    unsigned long long a = 1469598103934665603ULL, b = 1469598103934665603ULL;
    const unsigned long long P = 1099511628211ULL;
    for (int y = 0; y < (fr->height & ~1); y++) {
        const uint8_t* row = fr->data[0] + (ptrdiff_t)y * fr->linesize[0];
        for (int x = 0; x < (fr->width & ~1); x++) {
            unsigned v = depth > 8 ? ((const uint16_t*)row)[x] : row[x];
            unsigned v16 = (v << (16 - depth)) & 0xFFFF;
            unsigned v10 = v16 >> 6;
            unsigned t = (v16 >> 8) + ((v16 >> 7) & 1);
            unsigned v8 = t - (t >> 8);
            a = (a ^ (v10 & 0xff)) * P; a = (a ^ (v10 >> 8)) * P;
            b = (b ^ v8) * P;
        }
    }
    *h10 = a; *h8 = b;
}

// Decodes through IMFSourceReader with the given decoder output format and prints, per frame,
// the time and the same luma hashes as "avtest <file> hash" (h10 for 10-bit output, h8 for

static const GUID kFmtP010 = FOURCC_GUID(0x30313050);
static const GUID kFmtY210 = FOURCC_GUID(0x30313259);
static const GUID kFmtYUY2 = FOURCC_GUID(0x32595559);

// luma sample i of a row in each format, as a 16-bit MSB-aligned value
static inline unsigned LumaAt(const BYTE* row, int x, int fmt) {
    switch (fmt) {
    case 0: return row[x] << 8;                                   // NV12
    case 1: return ((const uint16_t*)row)[x];                     // P010
    case 2: return ((const uint16_t*)row)[(x & ~1) * 2 + (x & 1) * 2];   // Y210: Y0 U Y1 V
    default: return row[(x & ~1) * 2 + (x & 1) * 2] << 8;         // YUY2: Y0 U Y1 V
    }
}
static inline void ChromaAt(const BYTE* p, LONG pitch, int w, int h, int x, int y, int fmt, unsigned* u, unsi
gned* v) {
    if (fmt <= 1) {                                               // NV12 / P010: UV plane after Y
        const BYTE* uvRow = p + (size_t)pitch * h + (size_t)pitch * (y / 2);
        if (fmt == 0) { *u = uvRow[(x & ~1)] << 8; *v = uvRow[(x & ~1) + 1] << 8; }
        else { *u = ((const uint16_t*)uvRow)[x & ~1]; *v = ((const uint16_t*)uvRow)[(x & ~1) + 1]; }
    } else {
        const BYTE* row = p + (size_t)pitch * y;
        if (fmt == 2) { *u = ((const uint16_t*)row)[(x & ~1) * 2 + 1]; *v = ((const uint16_t*)row)[(x & ~1) *
 2 + 3]; }
        else { *u = row[(x & ~1) * 2 + 1] << 8; *v = row[(x & ~1) * 2 + 3] << 8; }
    }
    (void)w;
}

// BT.709 limited range -> BGRX, for a quick look at the picture.
static void SaveFrameBmp(const wchar_t* out, const BYTE* p, LONG pitch, int w, int h, int fmt) {
    BYTE* px = (BYTE*)malloc((size_t)w * h * 4);
    if (!px) return;
    for (int y = 0; y < h; y++) {
        const BYTE* row = p + (size_t)pitch * y;
        for (int x = 0; x < w; x++) {
            double Y = LumaAt(row, x, fmt) / 65535.0 * 255.0;
            unsigned uu, vv;
            ChromaAt(p, pitch, w, h, x, y, fmt, &uu, &vv);
            double U = uu / 65535.0 * 255.0 - 128.0, V = vv / 65535.0 * 255.0 - 128.0;
            double c = (Y - 16.0) * 255.0 / 219.0;
            double r = c + 1.5748 * 255.0 / 224.0 * V;
            double g = c - 0.1873 * 255.0 / 224.0 * U - 0.4681 * 255.0 / 224.0 * V;
            double b = c + 1.8556 * 255.0 / 224.0 * U;
            BYTE* d = px + ((size_t)y * w + x) * 4;
            d[0] = (BYTE)(b < 0 ? 0 : b > 255 ? 255 : b);
            d[1] = (BYTE)(g < 0 ? 0 : g > 255 ? 255 : g);
            d[2] = (BYTE)(r < 0 ? 0 : r > 255 ? 255 : r);
            d[3] = 0;
        }
    }
    SaveBmp(out, px, w, h);
    free(px);
}

Media Foundation 的程序，直接播放索尼相机拍摄的 **XAVC S / XAVC S-I**（H.264 High 4:2:2 10bit）视频。
不依赖旧版 Windows Media Player，不依赖第三方播放器，不需要微软签名或审批。

## 目标格式
## 支持的格式

| 格式 | 编码 | 现状 |
|------|------|------|
| XAVC HS | HEVC (H.265) Main 4:2:2 10 | **已可播放**（官方 HEVC 扩展已支持，视显卡而定可硬解） |
| XAVC S | H.264 High 4:2:2 Long GOP 10bit | 内置解码器不支持 → 本项目目标 |
| XAVC S-I | H.264 High 4:2:2 Intra 10bit | 内置解码器不支持 → 本项目目标 |
| 格式 | 编码 | 由谁解码 |
|------|------|----------|
| XAVC S（Long GOP） | H.264 High 4:2:2 10bit / High 10 4:2:0 | **本插件** |
| XAVC S-I（All-Intra） | H.264 High 4:2:2 Intra 10bit | **本插件** |
| XAVC S 8bit 4:2:0 | H.264 High | Windows 自带解码器（不受影响） |
| XAVC HS | HEVC 4:2:2 10bit | 微软 HEVC 扩展（不受影响） |

## 核心结论（见 `docs/feasibility.md`）
## 工作原理（详见 [`docs/feasibility.md`](docs/feasibility.md)）

- 系统内置 H.264 解码器只支持 8bit 4:2:0，无法解 XAVC S / S-I 的 High 4:2:2 (profile_idc=122)。
- H.264 4:2:2 在几乎所有消费级显卡上都**无法硬解**（仅 NVIDIA Blackwell/RTX50 例外），因此
  本插件以**多线程软解为主**，并在运行时探测可用的硬件解码、可用时优先走硬解。
- 解码输出采用 **Y210 (4:2:2 10bit)**，与已经能正常播放的 HEVC 路径一致，不降色度。
- **分发方式的生死问题**：新版播放器是 AppContainer 沙箱应用，是否加载"系统级注册的第三方

Windows 把 H.264 写死交给自带解码器，而它不支持 4:2:2 / 10bit。本插件不和它抢：

1. **字节流处理器**（每用户注册）用系统自带的 MP4 解析器打开文件，只把自带解码器解不了的
   H.264 视频流（profile 110/122/244/44）标记成私有格式 `XAVC`；
2. **解码器 MFT** 只处理 `XAVC` 格式，用 FFmpeg（libavcodec）多线程解码，输出 Y210 / P010 / YUY2 / NV12；
3. 普通 MP4 完全透传，仍由系统解码器处理。

已在真机验证：解码结果与 FFmpeg 逐帧比特一致（含拖动、结尾）；新版媒体播放器播放 S-I 4K24 实时 24fps；
4K60 Long GOP 在轻薄本 CPU 上自动降级（跳过 B 帧）以保持音画同步；资源管理器缩略图正常。

## 4. 解码器实现（`src/xavc_decoder.cpp`）

- **输入**：MPEG-4 源交给解码器的样本为 Annex-B（4 字节起始码，关键帧内带 SPS/PPS，每帧 8 个 slice），
  样本时间 = 显示时间（解码顺序送入，B 帧乱序）；`MF_MT_MPEG_SEQUENCE_HEADER` 也是 Annex-B。
  直接交给 libavcodec（帧级多线程），由它完成 B 帧重排。
- **输出格式**：在第一帧之前由 SPS 推出（色度格式、位深、裁剪后尺寸、VUI 色彩信息）。4:2:2 10bit 依次提供
  **Y210 → P010 → YUY2 → NV12**，渲染器取它的视频处理器支持的第一个。实测本机 Intel UHD 620 的
  D3D11 视频处理器不支持 Y210（支持 P010、YUY2），媒体播放器因此选 P010；支持 Y210 的显卡上保持完整 4:2:2。
- **DRAIN / FLUSH** 对应 avcodec 的 drain（之后 flush 以便继续接收输入）与 flush（拖动）。
- **FFmpeg 加载**：解码器第一次需要时，从本 DLL 所在目录 `LoadLibraryEx` 加载 avutil/avcodec（宿主进程
  不会搜索我们的目录）；字节流处理器完全不碰 FFmpeg。DLL 不依赖任何 MinGW 运行库（构建时检查）。
- **硬件解码**：H.264 4:2:2 只有 NVDEC（NVIDIA Blackwell 及以后）能解。存在 CUDA 设备时通过
  libavcodec 的 hwaccel 尝试，失败自动回退软件解码，并在本进程内记住结果。（本机无 NVIDIA 显卡，此路径未实测
。）

### 跟不上实时的处理（`IMFQualityAdvise2`）

逆向 + 实测：媒体引擎（新版媒体播放器）会 QI `IMFQualityAdvise2`，把渲染器的每个 `MEQualityNotify`
通过 `NotifyQualityEvent` 转发给解码器，**延迟量放在事件值里（VT_I8，100ns）**，自己从不调用
`SetDropMode` / `SetQualityLevel`，渲染器也不丢晚到的帧——不处理的话画面会越来越落后于声音（实测 4K60

| 自动级别 | 措施 |
|---|---|
| 1 | 非参考帧不做去块滤波 |
| 2 | 所有帧不做去块滤波 |
| 3 | 不解码非参考帧（索尼 Long GOP 的 B 帧） |
| 4 | 不解码任何 B 帧 |

H.264 High 4:2:2 在消费级 GPU 上几乎无硬解（NVIDIA 仅 Blackwell/RTX 50 新增），

On the 4K60 file: this i5-8265U can only decode about 30 fps at full quality. Media Player doesn't drop late frames by itself, so the video fell further and further behind the audio (5.5 s behind after 12 s). The decoder now handles this itself: it lowers quality step by step, skips B-frames, and jumps to the next keyframe if it is still behind. Once the CPU keeps up, it goes back up. A faster desktop CPU should play 4K60 at full frame rate.

Your two earlier questions
- 4:2:2 is not downsampled by the plugin. It offers 4:2:2 output (Y210) first. This laptop's Intel UHD 620 can't render that format, so Media Player picks 10-bit 4:2:0 (P010) instead. On graphics cards that support Y210, the full 4:2:2 is kept.
- Hardware decoding is detected automatically. Only NVIDIA RTX 50-series cards can hardware-decode H.264 4:2:2; on everything else it decodes on the CPU. This laptop has no NVIDIA card, so the hardware path is untested.
